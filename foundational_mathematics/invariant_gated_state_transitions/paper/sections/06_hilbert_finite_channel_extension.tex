\section{Hilbert representation corollaries}

The preceding section is the primary finite-to-infinite theorem spine.  This section retains the earlier Hilbert-space compression result as a downstream representation corollary.  It is useful when a finite adverse channel map is already known, but it does not replace the completion and faithfulness obligations of T16--T18.

\subsection{Finite-channel factorization corollary}

Let $\mathcal H$ be a Hilbert space, let $S=S^*\succeq0$ be bounded, and let
\[
V:\C^m\to\mathcal H
\]
be bounded with
\[
\operatorname{Ran}V\subseteq\operatorname{Ran}S^{1/2}.
\]
Let $C:\C^m\to(\ker S)^\perp$ be the unique minimum-norm source lift satisfying
\[
V=S^{1/2}C,
\]
and set
\[
B_C=C^*C\in M_m(\C).
\]

\begin{corollary}[Finite-channel Hilbert compression]
The following are equivalent:
\[
S-VV^*\succeq0,
\qquad
B_C\preceq I_m,
\qquad
\lambda_{\max}(B_C)\le1.
\]
If $\beta_C=\lambda_{\max}(B_C)<1$, then
\[
S-VV^*\succeq(1-\beta_C)S.
\]
More generally,
\[
S-VV^*\succeq\delta S
\iff
B_C\preceq(1-\delta)I_m
\qquad(0\le\delta\le1).
\]
\end{corollary}

\begin{proof}
The factorization
\[
S-VV^*=S^{1/2}(I-CC^*)S^{1/2}
\]
reduces the claim to contractivity of $C$.  The nonzero spectra of $CC^*$ and $C^*C$ agree.  The reserve statement follows from $CC^*\preceq\beta_C I$.
\end{proof}

\begin{corollary}[Strictly positive source form]
If $S\succeq cI$ for some $c>0$, then
\[
C=S^{-1/2}V,
\qquad
B_C=V^*S^{-1}V,
\]
so
\[
S-VV^*\succeq0
\iff
V^*S^{-1}V\preceq I_m.
\]
\end{corollary}

\begin{remark}[Classical matrix form]
The matrix $V^*S^{-1}V$ is the familiar finite Birman--Schwinger/Schur-complement form.  In the present paper it is downstream of source completion and target-faithful observer construction rather than the object from which the observer dimension is guessed.
\end{remark}

\begin{corollary}[Rank-one burden]
For $m=1$, $B_C=[\beta]$ and
\[
S-vv^*\succeq0
\iff
\beta\le1,
\]
with
\[
S-vv^*\succeq(1-\beta)S
\qquad(\beta<1).
\]
Thus the scalar decoder-burden inequality is the one-channel form of the finite obstruction theorem.
\end{corollary}

\begin{corollary}[Five-channel representation]
If T17--T18 determine a target-relevant memory rank $\nu=5$, then the terminal Hermitian certificate has size $5\times5$.  This dimension is forced by the recognized adverse range; it is not a reduction of the entire ambient carrier to five dimensions.
\end{corollary}

\subsection{Projection-defect corollary}

Let $P=P^*=P^2$ on a Hilbert space $\mathcal K$ and let $R:\C^m\to\mathcal K$ be bounded.  Define
\[
K_{\rm def}=R^*(I-P)R.
\]

\begin{corollary}[Positive projection defect]
One has
\[
0\preceq K_{\rm def}\preceq R^*R,
\]
\[
\ker K_{\rm def}
=\{c\in\C^m:Rc\in\operatorname{Ran}P\},
\]
and, whenever
\[
G_{\rm src}=R^*PR,
\qquad
G_{\rm tot}=R^*R,
\]
then
\[
G_{\rm tot}-G_{\rm src}=K_{\rm def}\succeq0.
\]
\end{corollary}

\begin{proof}
For every $c$,
\[
\langle c,K_{\rm def}c\rangle=\|(I-P)Rc\|^2.
\]
The order bound follows from $0\preceq I-P\preceq I$, and the Gram identity follows by subtraction.
\end{proof}

\begin{remark}[Hierarchy]
The logical order of the paper is now
\[
\text{positive form}
\to\text{completed carrier}
\to\text{bilateral flow and observer jets}
\to\text{finite-to-infinite completion}
\to\text{faithful minimal observer}
\to\text{finite Hermitian certificate}.
\]
The Hilbert formulas in this section are representation consequences of that hierarchy.
\end{remark}
